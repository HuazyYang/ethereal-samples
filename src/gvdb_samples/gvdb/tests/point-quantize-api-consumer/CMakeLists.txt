cmake_minimum_required(VERSION 3.24)
project(gvdb_point_quantize_installed_consumer LANGUAGES CXX)
include(CTest)
find_package(gvdbNvrhi 1 CONFIG REQUIRED)
add_executable(gvdb_point_quantize_api_consumer "${CMAKE_CURRENT_LIST_DIR}/../GVDBNvrhiPointQuantizeApiAbiTests.cpp")
target_link_libraries(gvdb_point_quantize_api_consumer PRIVATE gvdb::nvrhi_api)
add_test(NAME gvdb_point_quantize_api_consumer COMMAND gvdb_point_quantize_api_consumer)
option(GVDB_QUANTIZE_CONSUMER_CUDA_TEST "Run installed quantization with NVRHI CUDA" OFF)
set(consumer_targets gvdb_point_quantize_api_consumer)
if(GVDB_QUANTIZE_CONSUMER_CUDA_TEST)
    if(NOT TARGET nvrhi::cuda OR NOT EXISTS "${GVDB_OWNER_TEST_ARTIFACT}" OR NOT EXISTS "${GVDB_POINT_TEST_ARTIFACT}"
            OR NOT EXISTS "${GVDB_QUANTIZE_ENCODE_ARTIFACT}" OR NOT EXISTS "${GVDB_QUANTIZE_DECODE_ARTIFACT}")
        message(FATAL_ERROR "CUDA quantization consumer requires nvrhi::cuda and four offline artifacts")
    endif()
    add_executable(gvdb_point_quantize_cuda_consumer cuda.cpp)
    target_link_libraries(gvdb_point_quantize_cuda_consumer PRIVATE gvdb::nvrhi_api nvrhi::cuda)
    add_test(NAME gvdb_point_quantize_cuda_consumer COMMAND gvdb_point_quantize_cuda_consumer
        "${GVDB_OWNER_TEST_ARTIFACT}" "${GVDB_POINT_TEST_ARTIFACT}" "${GVDB_QUANTIZE_ENCODE_ARTIFACT}" "${GVDB_QUANTIZE_DECODE_ARTIFACT}")
    list(APPEND consumer_targets gvdb_point_quantize_cuda_consumer)
endif()
foreach(target IN LISTS consumer_targets)
    if(MSVC)
        target_compile_options(${target} PRIVATE /utf-8)
    endif()
    set_tests_properties(${target} PROPERTIES TIMEOUT 60
        ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:gvdb::nvrhi_api>;PATH=path_list_prepend:$<TARGET_FILE_DIR:nvrhi_core>;PATH=path_list_prepend:$<TARGET_FILE_DIR:nvrhi>")
endforeach()
