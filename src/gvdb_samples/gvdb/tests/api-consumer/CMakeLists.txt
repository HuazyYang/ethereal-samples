cmake_minimum_required(VERSION 3.24)
project(gvdb_nvrhi_installed_consumer LANGUAGES CXX)
include(CTest)
find_package(gvdbNvrhi 1 CONFIG REQUIRED)
add_executable(gvdb_api_consumer "${CMAKE_CURRENT_LIST_DIR}/../GVDBNvrhiApiAbiTests.cpp")
target_link_libraries(gvdb_api_consumer PRIVATE gvdb::nvrhi_api)
add_test(NAME gvdb_api_consumer COMMAND gvdb_api_consumer)
add_executable(gvdb_ownership_consumer ownership.cpp)
target_link_libraries(gvdb_ownership_consumer PRIVATE gvdb::nvrhi_volume)
add_test(NAME gvdb_ownership_consumer COMMAND gvdb_ownership_consumer)
if(MSVC)
    target_compile_options(gvdb_api_consumer PRIVATE /utf-8)
    target_compile_options(gvdb_ownership_consumer PRIVATE /utf-8)
endif()
set_tests_properties(gvdb_api_consumer gvdb_ownership_consumer PROPERTIES TIMEOUT 30
    ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:gvdb::nvrhi_api>;PATH=path_list_prepend:$<TARGET_FILE_DIR:nvrhi_core>;PATH=path_list_prepend:$<TARGET_FILE_DIR:nvrhi>")

option(GVDB_API_CONSUMER_CUDA_TEST "Run the installed API through NVRHI CUDA" OFF)
if(GVDB_API_CONSUMER_CUDA_TEST)
    if(NOT TARGET nvrhi::cuda OR NOT EXISTS "${GVDB_API_TEST_ARTIFACT}")
        message(FATAL_ERROR "CUDA consumer requires nvrhi::cuda and GVDB_API_TEST_ARTIFACT")
    endif()
    add_executable(gvdb_api_cuda_consumer cuda.cpp)
    target_link_libraries(gvdb_api_cuda_consumer PRIVATE gvdb::nvrhi_api nvrhi::cuda)
    if(MSVC)
        target_compile_options(gvdb_api_cuda_consumer PRIVATE /utf-8)
    endif()
    add_test(NAME gvdb_api_cuda_consumer COMMAND gvdb_api_cuda_consumer "${GVDB_API_TEST_ARTIFACT}")
    set_tests_properties(gvdb_api_cuda_consumer PROPERTIES TIMEOUT 60
        ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:gvdb::nvrhi_api>;PATH=path_list_prepend:$<TARGET_FILE_DIR:nvrhi_core>;PATH=path_list_prepend:$<TARGET_FILE_DIR:nvrhi>")
endif()
