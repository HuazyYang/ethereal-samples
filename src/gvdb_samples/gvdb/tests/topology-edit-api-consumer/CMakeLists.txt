cmake_minimum_required(VERSION 3.24)
project(gvdb_topology_edit_installed_consumer LANGUAGES CXX)
include(CTest)
find_package(gvdbNvrhi 1 CONFIG REQUIRED)
add_executable(gvdb_topology_edit_api_consumer ../GVDBNvrhiTopologyEditApiAbiTests.cpp)
target_link_libraries(gvdb_topology_edit_api_consumer PRIVATE gvdb::nvrhi_api)
add_test(NAME gvdb_topology_edit_api_consumer COMMAND gvdb_topology_edit_api_consumer)
option(GVDB_TOPOLOGY_EDIT_CONSUMER_CUDA_TEST "Run installed topology editing on CUDA" OFF)
set(consumer_targets gvdb_topology_edit_api_consumer)
if(GVDB_TOPOLOGY_EDIT_CONSUMER_CUDA_TEST)
    if(NOT TARGET nvrhi::cuda OR NOT EXISTS "${GVDB_OWNER_TEST_ARTIFACT}" OR NOT EXISTS "${GVDB_MARKER_TEST_ARTIFACT}")
        message(FATAL_ERROR "CUDA topology consumer needs offline owner/marker artifacts and nvrhi::cuda")
    endif()
    add_executable(gvdb_topology_edit_cuda_consumer cuda.cpp)
    target_link_libraries(gvdb_topology_edit_cuda_consumer PRIVATE gvdb::nvrhi_api nvrhi::cuda)
    add_test(NAME gvdb_topology_edit_cuda_consumer COMMAND gvdb_topology_edit_cuda_consumer
        "${GVDB_OWNER_TEST_ARTIFACT}" "${GVDB_MARKER_TEST_ARTIFACT}")
    list(APPEND consumer_targets gvdb_topology_edit_cuda_consumer)
endif()
foreach(target IN LISTS consumer_targets)
    if(MSVC)
        target_compile_options(${target} PRIVATE /utf-8)
    endif()
    if(WIN32)
        set(runtime_path PATH)
    else()
        set(runtime_path LD_LIBRARY_PATH)
    endif()
    set_tests_properties(${target} PROPERTIES TIMEOUT 60
        ENVIRONMENT_MODIFICATION "${runtime_path}=path_list_prepend:$<TARGET_FILE_DIR:gvdb::nvrhi_api>;${runtime_path}=path_list_prepend:$<TARGET_FILE_DIR:nvrhi_core>;${runtime_path}=path_list_prepend:$<TARGET_FILE_DIR:nvrhi>")
endforeach()
