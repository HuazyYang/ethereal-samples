cmake_minimum_required(VERSION 3.24)
project(gvdb_voxel_api_installed_consumer LANGUAGES CXX)
include(CTest)
find_package(gvdbNvrhi 1 CONFIG REQUIRED)
if(NOT TARGET gvdb::nvrhi_voxel_ops)
    message(FATAL_ERROR "The installed SDK was not built with GVDBNvrhiVoxelApi.cmake")
endif()
add_executable(gvdb_voxel_api_consumer "${CMAKE_CURRENT_LIST_DIR}/../GVDBNvrhiVoxelApiAbiTests.cpp")
target_link_libraries(gvdb_voxel_api_consumer PRIVATE gvdb::nvrhi_api)
add_test(NAME gvdb_voxel_api_consumer COMMAND gvdb_voxel_api_consumer)
add_executable(gvdb_voxel_core_consumer ownership.cpp)
target_link_libraries(gvdb_voxel_core_consumer PRIVATE gvdb::nvrhi_voxel_ops)
add_test(NAME gvdb_voxel_core_consumer COMMAND gvdb_voxel_core_consumer)
if(MSVC)
    target_compile_options(gvdb_voxel_api_consumer PRIVATE /utf-8)
    target_compile_options(gvdb_voxel_core_consumer PRIVATE /utf-8)
endif()
set_tests_properties(gvdb_voxel_api_consumer gvdb_voxel_core_consumer PROPERTIES TIMEOUT 30
    ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:gvdb::nvrhi_api>;PATH=path_list_prepend:$<TARGET_FILE_DIR:nvrhi_core>;PATH=path_list_prepend:$<TARGET_FILE_DIR:nvrhi>")
option(GVDB_VOXEL_CONSUMER_CUDA_TEST "Run installed voxel operations through NVRHI CUDA" OFF)
if(GVDB_VOXEL_CONSUMER_CUDA_TEST)
    if(NOT TARGET nvrhi::cuda OR NOT EXISTS "${GVDB_OWNER_TEST_ARTIFACT}" OR NOT EXISTS "${GVDB_VOXEL_TEST_ARTIFACT}")
        message(FATAL_ERROR "CUDA voxel consumer requires nvrhi::cuda and both offline artifacts")
    endif()
    add_executable(gvdb_voxel_cuda_consumer cuda.cpp)
    target_link_libraries(gvdb_voxel_cuda_consumer PRIVATE gvdb::nvrhi_api nvrhi::cuda)
    if(MSVC)
        target_compile_options(gvdb_voxel_cuda_consumer PRIVATE /utf-8)
    endif()
    add_test(NAME gvdb_voxel_cuda_consumer COMMAND gvdb_voxel_cuda_consumer "${GVDB_OWNER_TEST_ARTIFACT}" "${GVDB_VOXEL_TEST_ARTIFACT}")
    set_tests_properties(gvdb_voxel_cuda_consumer PROPERTIES TIMEOUT 60
        ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:gvdb::nvrhi_api>;PATH=path_list_prepend:$<TARGET_FILE_DIR:nvrhi_core>;PATH=path_list_prepend:$<TARGET_FILE_DIR:nvrhi>")
endif()
