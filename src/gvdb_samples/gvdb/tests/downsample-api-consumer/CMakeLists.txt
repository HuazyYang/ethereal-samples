cmake_minimum_required(VERSION 3.24)
project(gvdb_downsample_api_installed_consumer LANGUAGES CXX)
include(CTest)
find_package(gvdbNvrhi 1 CONFIG REQUIRED)
add_executable(gvdb_downsample_api_consumer "${CMAKE_CURRENT_LIST_DIR}/../GVDBNvrhiDownsampleApiAbiTests.cpp")
target_link_libraries(gvdb_downsample_api_consumer PRIVATE gvdb::nvrhi_api)
add_test(NAME gvdb_downsample_api_consumer COMMAND gvdb_downsample_api_consumer)
add_executable(gvdb_downsample_core_consumer ownership.cpp)
target_link_libraries(gvdb_downsample_core_consumer PRIVATE gvdb::nvrhi_downsample)
add_test(NAME gvdb_downsample_core_consumer COMMAND gvdb_downsample_core_consumer)
foreach(target gvdb_downsample_api_consumer gvdb_downsample_core_consumer)
    if(MSVC)
        target_compile_options(${target} PRIVATE /utf-8)
    endif()
    set_tests_properties(${target} PROPERTIES TIMEOUT 30 ENVIRONMENT_MODIFICATION
        "PATH=path_list_prepend:$<TARGET_FILE_DIR:gvdb::nvrhi_api>;PATH=path_list_prepend:$<TARGET_FILE_DIR:nvrhi_core>;PATH=path_list_prepend:$<TARGET_FILE_DIR:nvrhi>")
endforeach()
option(GVDB_DOWNSAMPLE_CONSUMER_CUDA_TEST "Run installed public downsample golden fixtures through NVRHI CUDA" OFF)
if(GVDB_DOWNSAMPLE_CONSUMER_CUDA_TEST)
    if(NOT TARGET nvrhi::cuda OR NOT EXISTS "${GVDB_DOWNSAMPLE_TEST_ARTIFACT}")
        message(FATAL_ERROR "CUDA downsample consumer requires nvrhi::cuda and downsample artifact")
    endif()
    add_executable(gvdb_downsample_cuda_consumer cuda.cpp)
    target_link_libraries(gvdb_downsample_cuda_consumer PRIVATE gvdb::nvrhi_api nvrhi::cuda)
    if(MSVC)
        target_compile_options(gvdb_downsample_cuda_consumer PRIVATE /utf-8)
    endif()
    add_test(NAME gvdb_downsample_cuda_consumer COMMAND gvdb_downsample_cuda_consumer "${GVDB_DOWNSAMPLE_TEST_ARTIFACT}")
    set_tests_properties(gvdb_downsample_cuda_consumer PROPERTIES TIMEOUT 60 ENVIRONMENT_MODIFICATION
        "PATH=path_list_prepend:$<TARGET_FILE_DIR:gvdb::nvrhi_api>;PATH=path_list_prepend:$<TARGET_FILE_DIR:nvrhi_core>;PATH=path_list_prepend:$<TARGET_FILE_DIR:nvrhi>")
endif()
