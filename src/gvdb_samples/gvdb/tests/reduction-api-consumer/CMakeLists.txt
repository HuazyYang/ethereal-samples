cmake_minimum_required(VERSION 3.24)
project(gvdb_reduction_api_installed_consumer LANGUAGES CXX)
include(CTest)
find_package(gvdbNvrhi 1 CONFIG REQUIRED)
add_executable(gvdb_reduction_api_consumer "${CMAKE_CURRENT_LIST_DIR}/../GVDBNvrhiReductionApiAbiTests.cpp")
target_link_libraries(gvdb_reduction_api_consumer PRIVATE gvdb::nvrhi_api)
add_test(NAME gvdb_reduction_api_consumer COMMAND gvdb_reduction_api_consumer)
if(MSVC)
    target_compile_options(gvdb_reduction_api_consumer PRIVATE /utf-8)
endif()
set_tests_properties(gvdb_reduction_api_consumer PROPERTIES TIMEOUT 30
    ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:gvdb::nvrhi_api>;PATH=path_list_prepend:$<TARGET_FILE_DIR:nvrhi_core>;PATH=path_list_prepend:$<TARGET_FILE_DIR:nvrhi>")
add_executable(gvdb_reduction_core_consumer ownership.cpp)
target_link_libraries(gvdb_reduction_core_consumer PRIVATE gvdb::nvrhi_reduction)
add_test(NAME gvdb_reduction_core_consumer COMMAND gvdb_reduction_core_consumer)
if(MSVC)
    target_compile_options(gvdb_reduction_core_consumer PRIVATE /utf-8)
endif()
set_tests_properties(gvdb_reduction_core_consumer PROPERTIES TIMEOUT 30
    ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:nvrhi_core>;PATH=path_list_prepend:$<TARGET_FILE_DIR:nvrhi>")
option(GVDB_REDUCTION_CONSUMER_CUDA_TEST "Run installed reduction scalar query through NVRHI CUDA" OFF)
if(GVDB_REDUCTION_CONSUMER_CUDA_TEST)
    if(NOT TARGET nvrhi::cuda OR NOT EXISTS "${GVDB_OWNER_TEST_ARTIFACT}" OR NOT EXISTS "${GVDB_REDUCTION_TEST_ARTIFACT}")
        message(FATAL_ERROR "CUDA reduction consumer requires nvrhi::cuda and both offline artifacts")
    endif()
    add_executable(gvdb_reduction_cuda_consumer cuda.cpp)
    target_link_libraries(gvdb_reduction_cuda_consumer PRIVATE gvdb::nvrhi_api nvrhi::cuda)
    if(MSVC)
        target_compile_options(gvdb_reduction_cuda_consumer PRIVATE /utf-8)
    endif()
    add_test(NAME gvdb_reduction_cuda_consumer COMMAND gvdb_reduction_cuda_consumer "${GVDB_OWNER_TEST_ARTIFACT}" "${GVDB_REDUCTION_TEST_ARTIFACT}")
    set_tests_properties(gvdb_reduction_cuda_consumer PROPERTIES TIMEOUT 60
        ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:gvdb::nvrhi_api>;PATH=path_list_prepend:$<TARGET_FILE_DIR:nvrhi_core>;PATH=path_list_prepend:$<TARGET_FILE_DIR:nvrhi>")
endif()
