cmake_minimum_required(VERSION 3.24)
project(gvdb_gather_api_installed_consumer LANGUAGES CXX)
include(CTest)
find_package(gvdbNvrhi 1 CONFIG REQUIRED)
add_executable(gvdb_gather_api_consumer "${CMAKE_CURRENT_LIST_DIR}/../GVDBNvrhiGatherApiAbiTests.cpp")
target_link_libraries(gvdb_gather_api_consumer PRIVATE gvdb::nvrhi_api)
add_test(NAME gvdb_gather_api_consumer COMMAND gvdb_gather_api_consumer)
if(MSVC)
    target_compile_options(gvdb_gather_api_consumer PRIVATE /utf-8)
endif()
set_tests_properties(gvdb_gather_api_consumer PROPERTIES TIMEOUT 30
    ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:gvdb::nvrhi_api>;PATH=path_list_prepend:$<TARGET_FILE_DIR:nvrhi_core>;PATH=path_list_prepend:$<TARGET_FILE_DIR:nvrhi>")
option(GVDB_GATHER_CONSUMER_CUDA_TEST "Run installed gather through NVRHI CUDA" OFF)
if(GVDB_GATHER_CONSUMER_CUDA_TEST)
    if(NOT TARGET nvrhi::cuda OR NOT EXISTS "${GVDB_OWNER_TEST_ARTIFACT}" OR NOT EXISTS "${GVDB_POINT_TEST_ARTIFACT}" OR NOT EXISTS "${GVDB_GATHER_TEST_ARTIFACT}")
        message(FATAL_ERROR "CUDA gather consumer requires nvrhi::cuda and owner/point/gather offline artifacts")
    endif()
    add_executable(gvdb_gather_cuda_consumer cuda.cpp)
    target_link_libraries(gvdb_gather_cuda_consumer PRIVATE gvdb::nvrhi_api nvrhi::cuda)
    if(MSVC)
        target_compile_options(gvdb_gather_cuda_consumer PRIVATE /utf-8)
    endif()
    set(weighted_arg "")
    if(DEFINED GVDB_INVERSE_DISTANCE_TEST_ARTIFACT)
        if(NOT EXISTS "${GVDB_INVERSE_DISTANCE_TEST_ARTIFACT}")
            message(FATAL_ERROR "Missing inverse-distance artifact")
        endif()
        set(weighted_arg "${GVDB_INVERSE_DISTANCE_TEST_ARTIFACT}")
    endif()
    add_test(NAME gvdb_gather_cuda_consumer COMMAND gvdb_gather_cuda_consumer "${GVDB_OWNER_TEST_ARTIFACT}" "${GVDB_POINT_TEST_ARTIFACT}" "${GVDB_GATHER_TEST_ARTIFACT}" ${weighted_arg})
    set_tests_properties(gvdb_gather_cuda_consumer PROPERTIES TIMEOUT 60
        ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:gvdb::nvrhi_api>;PATH=path_list_prepend:$<TARGET_FILE_DIR:nvrhi_core>;PATH=path_list_prepend:$<TARGET_FILE_DIR:nvrhi>")
endif()
