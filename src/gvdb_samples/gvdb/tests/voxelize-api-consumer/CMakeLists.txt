cmake_minimum_required(VERSION 3.24)
project(gvdb_voxelize_api_installed_consumer LANGUAGES CXX)
include(CTest)
find_package(gvdbNvrhi 1 CONFIG REQUIRED)
add_executable(gvdb_voxelize_api_consumer ../GVDBNvrhiVoxelizeApiAbiTests.cpp)
target_link_libraries(gvdb_voxelize_api_consumer PRIVATE gvdb::nvrhi_api)
add_test(NAME gvdb_voxelize_api_consumer COMMAND gvdb_voxelize_api_consumer)
option(GVDB_VOXELIZE_CONSUMER_CUDA_TEST "Test installed voxelizer through CUDA" OFF)
if(GVDB_VOXELIZE_CONSUMER_CUDA_TEST)
    if(NOT TARGET nvrhi::cuda OR NOT IS_DIRECTORY "${GVDB_VOXELIZE_ARTIFACT_DIRECTORY}")
        message(FATAL_ERROR "CUDA voxelizer consumer needs nvrhi::cuda and offline artifact directory")
    endif()
    add_executable(gvdb_voxelize_cuda_consumer cuda.cpp)
    target_link_libraries(gvdb_voxelize_cuda_consumer PRIVATE gvdb::nvrhi_api nvrhi::cuda)
    add_test(NAME gvdb_voxelize_cuda_consumer COMMAND gvdb_voxelize_cuda_consumer "${GVDB_VOXELIZE_ARTIFACT_DIRECTORY}")
endif()
foreach(target gvdb_voxelize_api_consumer gvdb_voxelize_cuda_consumer)
    if(TARGET ${target})
        if(MSVC)
            target_compile_options(${target} PRIVATE /utf-8)
        endif()
        set_tests_properties(${target} PROPERTIES TIMEOUT 60 ENVIRONMENT_MODIFICATION
            "PATH=path_list_prepend:$<TARGET_FILE_DIR:gvdb::nvrhi_api>;PATH=path_list_prepend:$<TARGET_FILE_DIR:nvrhi_core>;PATH=path_list_prepend:$<TARGET_FILE_DIR:nvrhi>")
    endif()
endforeach()
