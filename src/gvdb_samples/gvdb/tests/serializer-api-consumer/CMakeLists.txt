cmake_minimum_required(VERSION 3.24)
project(gvdb_serializer_installed_consumer LANGUAGES CXX)
include(CTest)
find_package(gvdbNvrhi 1 CONFIG REQUIRED)
add_executable(gvdb_serializer_abi_consumer "${CMAKE_CURRENT_LIST_DIR}/../GVDBNvrhiSerializerAbiTests.cpp")
target_link_libraries(gvdb_serializer_abi_consumer PRIVATE gvdb::nvrhi_api)
add_test(NAME gvdb_serializer_abi_consumer COMMAND gvdb_serializer_abi_consumer)
set_tests_properties(gvdb_serializer_abi_consumer PROPERTIES TIMEOUT 30
    ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:gvdb::nvrhi_api>;PATH=path_list_prepend:$<TARGET_FILE_DIR:nvrhi_core>;PATH=path_list_prepend:$<TARGET_FILE_DIR:nvrhi>")
option(GVDB_SERIALIZER_CONSUMER_CUDA_TEST "Run installed serialization through NVRHI CUDA" OFF)
if(GVDB_SERIALIZER_CONSUMER_CUDA_TEST)
    if(NOT TARGET nvrhi::cuda OR NOT EXISTS "${GVDB_OWNER_TEST_ARTIFACT}")
        message(FATAL_ERROR "Installed serializer GPU test requires nvrhi::cuda and the offline owner artifact")
    endif()
    add_executable(gvdb_serializer_cuda_consumer cuda.cpp)
    target_link_libraries(gvdb_serializer_cuda_consumer PRIVATE gvdb::nvrhi_api nvrhi::cuda)
    add_test(NAME gvdb_serializer_cuda_consumer COMMAND gvdb_serializer_cuda_consumer "${GVDB_OWNER_TEST_ARTIFACT}")
    set_tests_properties(gvdb_serializer_cuda_consumer PROPERTIES TIMEOUT 60
        ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:gvdb::nvrhi_api>;PATH=path_list_prepend:$<TARGET_FILE_DIR:nvrhi_core>;PATH=path_list_prepend:$<TARGET_FILE_DIR:nvrhi>")
endif()
