if(ETHEREAL_WITH_SLANG AND TARGET gvdb_scene_rt)
    include(../sample-utils/SceneAssets.cmake)
    include(../sample-utils/OwnedVolumeRaycast.cmake)
    add_executable(interactive-optix main.cpp InteractiveScene.cpp InteractiveTests.cpp)
    target_link_libraries(interactive-optix PRIVATE gvdb_scene_rt gvdb_scene_assets gvdb_volume_viewer gvdb_owned_raycast)
    target_compile_definitions(interactive-optix PRIVATE "GVDB_SAMPLES_ASSETS_DIR=\"${CMAKE_CURRENT_SOURCE_DIR}/../assets\"")
    ethereal_use_agility_sdk(interactive-optix)
    if(NVRHI_WITH_OPTIX)
        target_link_libraries(interactive-optix PRIVATE nvrhi::cuda)
        target_compile_definitions(interactive-optix PRIVATE ETHEREAL_TEST_CUDA=1)
    endif()
    foreach(api DX12 VULKAN CUDA)
        if(NVRHI_WITH_${api} AND (NOT api STREQUAL "CUDA" OR NVRHI_WITH_OPTIX))
            if(api STREQUAL "DX12")
                set(argument -dx12)
            elseif(api STREQUAL "VULKAN")
                set(argument -vk)
            else()
                set(argument -cuda)
            endif()
            add_test(NAME gvdb_interactive_optix_analytic_${api} COMMAND interactive-optix ${argument} --self-test)
            set_tests_properties(gvdb_interactive_optix_analytic_${api} PROPERTIES TIMEOUT 180 RUN_SERIAL TRUE)
            foreach(mode volume trilinear emptyskip levelset)
                add_test(NAME gvdb_interactive_optix_asset_${api}_${mode} COMMAND interactive-optix ${argument}
                    --headless --shading ${mode} --samples 2 --width 128 --height 96 --verify
                    --output "${CMAKE_CURRENT_BINARY_DIR}/interactive-${api}-${mode}.png")
                set_tests_properties(gvdb_interactive_optix_asset_${api}_${mode} PROPERTIES TIMEOUT 240 RUN_SERIAL TRUE)
            endforeach()
        endif()
    endforeach()
endif()
