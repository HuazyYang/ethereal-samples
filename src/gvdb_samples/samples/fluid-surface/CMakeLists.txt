if(ETHEREAL_WITH_SLANG AND TARGET gvdb_slang_runtime)
    include(../sample-utils/SceneAssets.cmake)
    include(../sample-utils/OwnedVolumeSurface.cmake)
    include(../../gvdb/GVDBNvrhiGatherApi.cmake)
    include(../../gvdb/GVDBNvrhiApronApi.cmake)
    set(fluid_owned_config "")
    foreach(pair "GVDBAtlasRepack relocateAtlas" "GVDBPointStreamConversion convertPointStream" "GVDBAtlasGather gatherOwnedAtlas" "GVDBAtlasMutation mutateAtlas")
        separate_arguments(parts UNIX_COMMAND "${pair}")
        list(GET parts 0 module)
        list(GET parts 1 entry)
        string(APPEND fluid_owned_config "\"${CMAKE_CURRENT_SOURCE_DIR}/../../gvdb/slang/${module}.slang\" -T cs -E ${entry}\n")
    endforeach()
    file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/fluid-owned.cfg" CONTENT "${fluid_owned_config}" @ONLY)
    ethereal_compile_shaders(TARGET gvdb_fluid_owned_shaders SLANG CUDA ALL_PLATFORMS
        CONFIG "${CMAKE_CURRENT_BINARY_DIR}/fluid-owned.cfg"
        OUTPUT_DIRECTORY "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/shaders/gvdb_fluid_owned")
    file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/fluid-surface.cfg" CONTENT
        "\"${CMAKE_CURRENT_SOURCE_DIR}/FluidSurface.slang\" -T cs -E publishFluidAtlas\n" @ONLY)
    ethereal_compile_shaders(TARGET gvdb_fluid_surface_shaders SLANG CUDA ALL_PLATFORMS
        CONFIG "${CMAKE_CURRENT_BINARY_DIR}/fluid-surface.cfg"
        OUTPUT_DIRECTORY "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/shaders/gvdb_slang")
    add_executable(fluid-surface main.cpp PortableFluid.cpp FluidScene.cpp FluidTests.cpp)
    target_link_libraries(fluid-surface PRIVATE gvdb_slang_runtime gvdb_volume_viewer gvdb_scene_assets gvdb_owned_surface gvdb::nvrhi_api)
    target_compile_definitions(fluid-surface PRIVATE "GVDB_SAMPLES_ASSETS_DIR=\"${CMAKE_CURRENT_SOURCE_DIR}/../assets\"")
    add_dependencies(fluid-surface gvdb_fluid_owned_shaders gvdb_fluid_surface_shaders gvdb_slang_shaders)
    if(TARGET gvdb_scene_rt)
        target_link_libraries(fluid-surface PRIVATE gvdb_scene_rt)
        target_compile_definitions(fluid-surface PRIVATE ETHEREAL_FLUID_RT=1)
    else()
        target_compile_definitions(fluid-surface PRIVATE ETHEREAL_FLUID_RT=0)
    endif()
    ethereal_use_agility_sdk(fluid-surface)
    if(NVRHI_WITH_CUDA)
        target_link_libraries(fluid-surface PRIVATE nvrhi::cuda)
        target_compile_definitions(fluid-surface PRIVATE ETHEREAL_TEST_CUDA=1)
    endif()
    add_test(NAME gvdb_fluid_topology_cpu COMMAND fluid-surface --topology-test)
    set_tests_properties(gvdb_fluid_topology_cpu PROPERTIES TIMEOUT 60)
    foreach(api DX11 DX12 VULKAN CUDA)
        if(NVRHI_WITH_${api})
            if(api STREQUAL "VULKAN")
                set(argument -vk)
            elseif(api STREQUAL "DX11")
                set(argument -dx11)
            elseif(api STREQUAL "DX12")
                set(argument -dx12)
            else()
                set(argument -cuda)
            endif()
            add_test(NAME gvdb_fluid_analytic_${api} COMMAND fluid-surface ${argument} --renderer compute --self-test)
            add_test(NAME gvdb_fluid_compute_${api} COMMAND fluid-surface ${argument} --renderer compute --headless --test-scene
                --frames 2 --samples 2 --width 96 --height 72 --output "${CMAKE_CURRENT_BINARY_DIR}/fluid-compute-${api}.png")
            set_tests_properties(gvdb_fluid_analytic_${api} gvdb_fluid_compute_${api} PROPERTIES TIMEOUT 240 RUN_SERIAL TRUE)
            if(TARGET gvdb_scene_rt AND NOT api STREQUAL "DX11" AND (NOT api STREQUAL "CUDA" OR NVRHI_WITH_OPTIX))
                add_test(NAME gvdb_fluid_rt_analytic_${api} COMMAND fluid-surface ${argument} --self-test)
                add_test(NAME gvdb_fluid_rt_${api} COMMAND fluid-surface ${argument} --headless --test-scene --verify-volume
                    --frames 2 --samples 2 --width 96 --height 72 --output "${CMAKE_CURRENT_BINARY_DIR}/fluid-rt-${api}.png")
                add_test(NAME gvdb_fluid_full_${api} COMMAND fluid-surface ${argument} --headless --verify-volume
                    --frames 1 --samples 1 --width 96 --height 72 --output "${CMAKE_CURRENT_BINARY_DIR}/fluid-full-${api}.png")
                set_tests_properties(gvdb_fluid_rt_analytic_${api} gvdb_fluid_rt_${api} PROPERTIES TIMEOUT 240 RUN_SERIAL TRUE)
                set_tests_properties(gvdb_fluid_full_${api} PROPERTIES TIMEOUT 600 RUN_SERIAL TRUE)
            elseif(api STREQUAL "DX11")
                add_test(NAME gvdb_fluid_full_${api} COMMAND fluid-surface ${argument} --renderer compute --headless
                    --frames 1 --samples 1 --width 96 --height 72 --output "${CMAKE_CURRENT_BINARY_DIR}/fluid-full-${api}.png")
                set_tests_properties(gvdb_fluid_full_${api} PROPERTIES TIMEOUT 600 RUN_SERIAL TRUE)
            endif()
        endif()
    endforeach()
else()
    gvdb_add_sample(fluid-surface OPTIX SOURCES legacy-main.cpp FluidSystem.h FluidSystem.cpp)
    if(TARGET fluid-surface)
        gvdb_add_ptx(fluid-surface_ptx SOURCES FluidSystemKernels.cu)
        target_sources(fluid-surface_ptx_objects PRIVATE FluidSystem.h)
        add_dependencies(fluid-surface fluid-surface_ptx)
    endif()
endif()
