if(ETHEREAL_WITH_SLANG AND ETHEREAL_BUILD_GVDB)
    include("${CMAKE_CURRENT_SOURCE_DIR}/../sample-utils/OwnedVolumeRaycast.cmake")
    file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/render-kernel.cfg"
        CONTENT "\"${CMAKE_CURRENT_SOURCE_DIR}/RenderCustom.slang\" -T cs -E raycast_kernel\n" @ONLY)
    ethereal_compile_shaders(TARGET gvdb_render_kernel_shaders SLANG CUDA ALL_PLATFORMS
        CONFIG "${CMAKE_CURRENT_BINARY_DIR}/render-kernel.cfg"
        OUTPUT_DIRECTORY "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/shaders/render_kernel"
        INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}/../../gvdb/slang")
    add_executable(render-kernel main.cpp RenderCustom.cpp)
    target_link_libraries(render-kernel PRIVATE gvdb_owned_raycast donut_app stb)
    target_include_directories(render-kernel PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../..")
    target_compile_definitions(render-kernel PRIVATE "GVDB_SAMPLES_ASSETS_DIR=\"${CMAKE_CURRENT_SOURCE_DIR}/../assets\"")
    add_dependencies(render-kernel gvdb_render_kernel_shaders)
    ethereal_use_agility_sdk(render-kernel)
    if(NVRHI_WITH_CUDA)
        target_link_libraries(render-kernel PRIVATE nvrhi::cuda)
        target_compile_definitions(render-kernel PRIVATE ETHEREAL_TEST_CUDA=1)
    endif()
    if(BUILD_TESTING)
        file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/render-kernel-reference.cfg"
            CONTENT "\"${CMAKE_CURRENT_SOURCE_DIR}/RenderCustomTyped.slang\" -T cs -E raycast_kernel\n" @ONLY)
        ethereal_compile_shaders(TARGET gvdb_render_kernel_reference_shaders SLANG CUDA ALL_PLATFORMS
            CONFIG "${CMAKE_CURRENT_BINARY_DIR}/render-kernel-reference.cfg"
            OUTPUT_DIRECTORY "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/shaders/render_kernel"
            INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}/../../gvdb/slang")
        add_executable(gvdb_render_kernel_tests RenderCustomTests.cpp RenderCustom.cpp)
        target_link_libraries(gvdb_render_kernel_tests PRIVATE gvdb_owned_raycast donut_app)
        target_include_directories(gvdb_render_kernel_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../..")
        target_compile_definitions(gvdb_render_kernel_tests PRIVATE "GVDB_SAMPLES_ASSETS_DIR=\"${CMAKE_CURRENT_SOURCE_DIR}/../assets\"")
        add_dependencies(gvdb_render_kernel_tests gvdb_render_kernel_shaders gvdb_render_kernel_reference_shaders)
        ethereal_use_agility_sdk(gvdb_render_kernel_tests)
        if(MSVC)
            target_compile_options(gvdb_render_kernel_tests PRIVATE /utf-8)
        endif()
        if(NVRHI_WITH_CUDA)
            target_link_libraries(gvdb_render_kernel_tests PRIVATE nvrhi::cuda)
            target_compile_definitions(gvdb_render_kernel_tests PRIVATE ETHEREAL_TEST_CUDA=1)
        endif()
    endif()
    foreach(api DX11 DX12 VULKAN CUDA)
        if(NVRHI_WITH_${api})
            if(api STREQUAL "VULKAN")
                set(argument -vk)
            elseif(api STREQUAL "DX11")
                set(argument -dx11)
            elseif(api STREQUAL "DX12")
                set(argument -dx12)
            else()
                set(argument -cuda)
            endif()
            add_test(NAME gvdb_render_kernel_${api} COMMAND render-kernel ${argument}
                --width 128 --height 96 --step 1 --output "${CMAKE_CURRENT_BINARY_DIR}/custom-${api}.png")
            add_test(NAME gvdb_render_kernel_analytic_${api} COMMAND render-kernel ${argument} --self-test)
            if(BUILD_TESTING)
                add_test(NAME gvdb_render_kernel_owned_${api} COMMAND gvdb_render_kernel_tests ${argument})
                set_tests_properties(gvdb_render_kernel_owned_${api} PROPERTIES TIMEOUT 180)
            endif()
            set_tests_properties(gvdb_render_kernel_${api} gvdb_render_kernel_analytic_${api} PROPERTIES TIMEOUT 180)
        endif()
    endforeach()
endif()
