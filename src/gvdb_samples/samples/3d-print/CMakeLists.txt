if(ETHEREAL_WITH_SLANG AND ETHEREAL_BUILD_GVDB)
    include("${CMAKE_CURRENT_SOURCE_DIR}/../../gvdb/GVDBNvrhiResampleApi.cmake")
    include("${CMAKE_CURRENT_SOURCE_DIR}/../sample-utils/OwnedVolumeView.cmake")
    add_executable(3d-print main.cpp PrintVolume.cpp PrintTests.cpp ../sample-utils/ObjLoader.cpp)
    target_link_libraries(3d-print PRIVATE gvdb_volume_viewer gvdb::nvrhi_api)
    target_include_directories(3d-print PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../.." "${CMAKE_CURRENT_SOURCE_DIR}/..")
    target_compile_definitions(3d-print PRIVATE "GVDB_SAMPLES_ASSETS_DIR=\"${CMAKE_CURRENT_SOURCE_DIR}/../assets\"")
    add_dependencies(3d-print gvdb_slang_shaders gvdb_owned_raycast_shaders gvdb_nvrhi_resample_shaders)
    ethereal_use_agility_sdk(3d-print)
    if(NVRHI_WITH_CUDA)
        target_link_libraries(3d-print PRIVATE nvrhi::cuda)
        target_compile_definitions(3d-print PRIVATE ETHEREAL_TEST_CUDA=1)
    endif()
    foreach(api DX11 DX12 VULKAN CUDA)
        if(NVRHI_WITH_${api})
            if(api STREQUAL "VULKAN")
                set(argument -vk)
            elseif(api STREQUAL "DX11")
                set(argument -dx11)
            elseif(api STREQUAL "DX12")
                set(argument -dx12)
            else()
                set(argument -cuda)
            endif()
            add_test(NAME gvdb_print_analytic_${api} COMMAND 3d-print ${argument} --self-test)
            add_test(NAME gvdb_print_lucy_${api} COMMAND 3d-print ${argument} --headless --verify --voxel-size .5
                --width 128 --height 96 --output "${CMAKE_CURRENT_BINARY_DIR}/lucy-${api}.png")
            set_tests_properties(gvdb_print_analytic_${api} gvdb_print_lucy_${api} PROPERTIES TIMEOUT 240)
        endif()
    endforeach()
else()
    gvdb_add_sample(3d-print SOURCES legacy-main.cpp)
endif()
