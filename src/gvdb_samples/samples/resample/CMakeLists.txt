if(ETHEREAL_WITH_SLANG AND ETHEREAL_BUILD_GVDB)
    add_executable(resample main.cpp ResampleVolume.cpp ResampleTests.cpp)
    target_link_libraries(resample PRIVATE gvdb_volume_viewer)
    target_include_directories(resample PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../..")
    target_compile_definitions(resample PRIVATE "GVDB_SAMPLES_ASSETS_DIR=\"${CMAKE_CURRENT_SOURCE_DIR}/../assets\"")
    add_dependencies(resample gvdb_slang_shaders)
    ethereal_use_agility_sdk(resample)
    if(NVRHI_WITH_CUDA)
        target_link_libraries(resample PRIVATE nvrhi::cuda)
        target_compile_definitions(resample PRIVATE ETHEREAL_TEST_CUDA=1)
    endif()
    foreach(api DX11 DX12 VULKAN CUDA)
        if(NVRHI_WITH_${api})
            if(api STREQUAL "VULKAN")
                set(argument -vk)
            elseif(api STREQUAL "DX11")
                set(argument -dx11)
            elseif(api STREQUAL "DX12")
                set(argument -dx12)
            else()
                set(argument -cuda)
            endif()
            add_test(NAME gvdb_resample_analytic_${api} COMMAND resample ${argument} --self-test)
            add_test(NAME gvdb_resample_head_${api} COMMAND resample ${argument} --headless --verify --sparse
                --width 128 --height 96 --step 1 --output "${CMAKE_CURRENT_BINARY_DIR}/head-${api}.png")
            set_tests_properties(gvdb_resample_analytic_${api} gvdb_resample_head_${api} PROPERTIES TIMEOUT 240)
        endif()
    endforeach()
else()
    gvdb_add_sample(resample SOURCES legacy-main.cpp)
endif()
