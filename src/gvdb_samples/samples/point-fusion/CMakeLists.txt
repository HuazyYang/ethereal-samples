if(ETHEREAL_WITH_SLANG AND TARGET gvdb_slang_runtime)
    if(NOT TARGET gvdb_point_fusion)
        include(PointFusionTests.cmake)
    endif()
    add_executable(point-fusion main.cpp PointFusionScene.cpp PointFusionTests.cpp)
    target_link_libraries(point-fusion PRIVATE gvdb_point_fusion gvdb_volume_viewer)
    ethereal_use_agility_sdk(point-fusion)
    if(NVRHI_WITH_CUDA)
        target_link_libraries(point-fusion PRIVATE nvrhi::cuda)
        target_compile_definitions(point-fusion PRIVATE ETHEREAL_TEST_CUDA=1)
    endif()
    foreach(api DX11 DX12 VULKAN CUDA)
        if(NVRHI_WITH_${api})
            if(api STREQUAL "VULKAN")
                set(argument -vk)
            elseif(api STREQUAL "DX11")
                set(argument -dx11)
            elseif(api STREQUAL "DX12")
                set(argument -dx12)
            else()
                set(argument -cuda)
            endif()
            add_test(NAME gvdb_point_fusion_input_${api} COMMAND point-fusion ${argument} --input-test)
            add_test(NAME gvdb_point_fusion_city_${api} COMMAND point-fusion ${argument} --headless --frames 2 --color --speed 1
                --width 128 --height 96 --output "${CMAKE_CURRENT_BINARY_DIR}/city-${api}.png")
            set_tests_properties(gvdb_point_fusion_input_${api} gvdb_point_fusion_city_${api} PROPERTIES TIMEOUT 240)
        endif()
    endforeach()
else()
    gvdb_add_sample(point-fusion SOURCES legacy-main.cpp PTX PointFusionScan.cu)
endif()
