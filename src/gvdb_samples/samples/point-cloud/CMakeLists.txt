if(ETHEREAL_WITH_SLANG AND TARGET gvdb_slang_runtime)
    if(NOT TARGET gvdb_point_cloud_reconstruction)
        include(PointCloudTests.cmake)
    endif()
    include("${CMAKE_CURRENT_LIST_DIR}/../sample-utils/OwnedVolumeSurface.cmake")
    add_executable(point-cloud main.cpp PointCloudAppScene.cpp PointCloudAppTests.cpp ../sample-utils/ObjLoader.cpp)
    target_link_libraries(point-cloud PRIVATE gvdb_point_cloud_reconstruction gvdb_volume_viewer gvdb_owned_surface)
    target_compile_definitions(point-cloud PRIVATE "GVDB_SAMPLES_ASSETS_DIR=\"${CMAKE_CURRENT_SOURCE_DIR}/../assets\"")
    if(TARGET gvdb_scene_rt)
        target_link_libraries(point-cloud PRIVATE gvdb_scene_rt)
        target_compile_definitions(point-cloud PRIVATE ETHEREAL_POINT_CLOUD_RT=1)
    endif()
    ethereal_use_agility_sdk(point-cloud)
    if(NVRHI_WITH_CUDA)
        target_link_libraries(point-cloud PRIVATE nvrhi::cuda)
        target_compile_definitions(point-cloud PRIVATE ETHEREAL_TEST_CUDA=1)
    endif()
    foreach(api DX11 DX12 VULKAN CUDA)
        if(NVRHI_WITH_${api})
            if(api STREQUAL "VULKAN")
                set(argument -vk)
            elseif(api STREQUAL "DX11")
                set(argument -dx11)
            elseif(api STREQUAL "DX12")
                set(argument -dx12)
            else()
                set(argument -cuda)
            endif()
            add_test(NAME gvdb_point_cloud_app_input_${api} COMMAND point-cloud ${argument} --input-test)
            add_test(NAME gvdb_point_cloud_compute_${api} COMMAND point-cloud ${argument} --renderer compute --headless
                --frames 2 --width 128 --height 96 --output "${CMAKE_CURRENT_BINARY_DIR}/point-cloud-compute-${api}.png")
            set_tests_properties(gvdb_point_cloud_compute_${api} PROPERTIES TIMEOUT 240 RUN_SERIAL TRUE)
            if(TARGET gvdb_scene_rt AND NOT api STREQUAL "DX11" AND (NOT api STREQUAL "CUDA" OR NVRHI_WITH_OPTIX))
                add_test(NAME gvdb_point_cloud_rt_clip_${api} COMMAND point-cloud ${argument} --ray-policy-test)
                set_tests_properties(gvdb_point_cloud_rt_clip_${api} PROPERTIES TIMEOUT 120 RUN_SERIAL TRUE)
                add_test(NAME gvdb_point_cloud_rt_${api} COMMAND point-cloud ${argument} --headless --samples 2 --verify-volume
                    --width 128 --height 96 --output "${CMAKE_CURRENT_BINARY_DIR}/point-cloud-rt-${api}.png")
                set_tests_properties(gvdb_point_cloud_rt_${api} PROPERTIES TIMEOUT 240 RUN_SERIAL TRUE)
            endif()
        endif()
    endforeach()
else()
    gvdb_add_sample(point-cloud OPTIX SOURCES legacy-main.cpp SceneFile.h SceneFile.cpp)
endif()
