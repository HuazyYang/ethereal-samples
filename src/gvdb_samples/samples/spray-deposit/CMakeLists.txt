if(ETHEREAL_WITH_SLANG AND TARGET gvdb_slang_runtime)
    if(NOT TARGET gvdb_spray_simulation)
        include(SprayTests.cmake)
    endif()
    add_executable(spray-deposit main.cpp SprayScene.cpp SprayAppTests.cpp)
    target_link_libraries(spray-deposit PRIVATE gvdb_spray_simulation gvdb_volume_viewer)
    target_compile_definitions(spray-deposit PRIVATE "GVDB_SAMPLES_ASSETS_DIR=\"${CMAKE_CURRENT_SOURCE_DIR}/../assets\"")
    ethereal_use_agility_sdk(spray-deposit)
    if(NVRHI_WITH_CUDA)
        target_link_libraries(spray-deposit PRIVATE nvrhi::cuda)
        target_compile_definitions(spray-deposit PRIVATE ETHEREAL_TEST_CUDA=1)
    endif()
    foreach(api DX11 DX12 VULKAN CUDA)
        if(NVRHI_WITH_${api})
            if(api STREQUAL "VULKAN")
                set(argument -vk)
            elseif(api STREQUAL "DX11")
                set(argument -dx11)
            elseif(api STREQUAL "DX12")
                set(argument -dx12)
            else()
                set(argument -cuda)
            endif()
            add_test(NAME gvdb_spray_controls_${api} COMMAND spray-deposit ${argument} --self-test)
            set_tests_properties(gvdb_spray_controls_${api} PROPERTIES TIMEOUT 240 RUN_SERIAL TRUE)
            add_test(NAME gvdb_spray_application_${api} COMMAND spray-deposit ${argument} --headless --frames 3
                --width 128 --height 96 --output "${CMAKE_CURRENT_BINARY_DIR}/spray-${api}.png")
            set_tests_properties(gvdb_spray_application_${api} PROPERTIES TIMEOUT 240 RUN_SERIAL TRUE)
            if(NOT api STREQUAL "CUDA")
                add_test(NAME gvdb_spray_window_${api} COMMAND spray-deposit ${argument} --validation --gui-input-test --frames 2
                    --width 640 --height 480 --screenshot "${CMAKE_CURRENT_BINARY_DIR}/spray-window-${api}.png")
                set_tests_properties(gvdb_spray_window_${api} PROPERTIES TIMEOUT 240 RUN_SERIAL TRUE)
            endif()
        endif()
    endforeach()
else()
    gvdb_add_sample(spray-deposit SOURCES legacy-main.cpp)
endif()
