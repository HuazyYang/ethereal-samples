if(ETHEREAL_WITH_SLANG AND ETHEREAL_BUILD_GVDB)
    include("${CMAKE_CURRENT_SOURCE_DIR}/../sample-utils/OwnedVolumeRaycast.cmake")
    include("${CMAKE_CURRENT_SOURCE_DIR}/../sample-utils/OwnedVolumeRaycastTests.cmake")
    include("${CMAKE_CURRENT_SOURCE_DIR}/../sample-utils/OwnedVolumeTricubic.cmake")
    add_executable(render-to-file main.cpp)
    target_link_libraries(render-to-file PRIVATE gvdb_owned_raycast gvdb_owned_tricubic donut_app stb)
    target_include_directories(render-to-file PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../..")
    target_compile_definitions(render-to-file PRIVATE "GVDB_SAMPLES_ASSETS_DIR=\"${CMAKE_CURRENT_SOURCE_DIR}/../assets\"")
    add_dependencies(render-to-file gvdb_owned_raycast_shaders)
    ethereal_use_agility_sdk(render-to-file)
    if(NVRHI_WITH_CUDA)
        target_link_libraries(render-to-file PRIVATE nvrhi::cuda)
        target_compile_definitions(render-to-file PRIVATE ETHEREAL_TEST_CUDA=1)
    endif()
    foreach(api DX11 DX12 VULKAN CUDA)
        if(NVRHI_WITH_${api})
            if(api STREQUAL "VULKAN")
                set(argument -vk)
            elseif(api STREQUAL "DX11")
                set(argument -dx11)
            elseif(api STREQUAL "DX12")
                set(argument -dx12)
            else()
                set(argument -cuda)
            endif()
            add_test(NAME gvdb_render_file_${api} COMMAND render-to-file ${argument}
                --width 128 --height 96 --step 1 --output "${CMAKE_CURRENT_BINARY_DIR}/volume-${api}.png")
            set_tests_properties(gvdb_render_file_${api} PROPERTIES TIMEOUT 180)
            add_test(NAME gvdb_render_tricubic_${api} COMMAND render-to-file ${argument}
                --tricubic --width 64 --height 48 --step 1 --output "${CMAKE_CURRENT_BINARY_DIR}/tricubic-${api}.png")
            set_tests_properties(gvdb_render_tricubic_${api} PROPERTIES TIMEOUT 180 RUN_SERIAL TRUE)
        endif()
    endforeach()
else()
    gvdb_add_sample(render-to-file SOURCES legacy-main.cpp)
endif()
